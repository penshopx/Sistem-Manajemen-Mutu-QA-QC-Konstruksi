## 2. Service Layer untuk Sinkronisasi

```typescript
// artifacts/api-server/src/services/mir-batch-sync.ts
import { db } from '../lib/db';
import { materialBatches, inspections, auditLogs } from '../../lib/db/schema';
import { eq, and } from 'drizzle-orm';

export type MIRResult = 'approved' | 'rejected' | 'conditional';

export class MIRBatchSyncError extends Error {
  constructor(message: string, public code: string) {
    super(message);
  }
}

/**
 * Sinkronkan hasil MIR ke status batch material dalam satu transaksi atomik.
 * 
 * @throws MIRBatchSyncError jika batch tidak ditemukan atau tidak dalam proyek yang sama
 */
export async function syncMIRResultToBatch(params: {
  inspectionId: number;
  projectId: string;
  result: MIRResult;
  userId: string;
  notes?: string;
}): Promise<void> {
  const { inspectionId, projectId, result, userId, notes } = params;

  await db.transaction(async (tx) => {
    // 1. Verifikasi inspeksi adalah MIR dan dalam proyek yang sama
    const inspection = await tx.query.inspections.findFirst({
      where: and(
        eq(inspections.id, inspectionId),
        eq(inspections.projectId, projectId),
        eq(inspections.type, 'MATERIAL_INSPECTION')
      ),
    });

    if (!inspection) {
      throw new MIRBatchSyncError(
        'Inspeksi tidak ditemukan atau bukan MIR',
        'INSPECTION_NOT_FOUND'
      );
    }

    if (!inspection.relatedBatchId) {
      throw new MIRBatchSyncError(
        'MIR tidak tertaut ke batch material',
        'BATCH_NOT_LINKED'
      );
    }

    // 2. Verifikasi batch dalam proyek yang sama
    const batch = await tx.query.materialBatches.findFirst({
      where: and(
        eq(materialBatches.id, inspection.relatedBatchId),
        eq(materialBatches.projectId, projectId)
      ),
    });

    if (!batch) {
      throw new MIRBatchSyncError(
        'Batch material tidak ditemukan dalam proyek ini',
        'PROJECT_ISOLATION_VIOLATION'
      );
    }

    // 3. Baca status sebelum perubahan
    const previousStatus = batch.inspectionStatus;

    // 4. Update status batch
    const [updatedBatch] = await tx.update(materialBatches)
      .set({
        inspectionStatus: result,
        updatedAt: new Date(),
      })
      .where(eq(materialBatches.id, batch.id))
      .returning();

    // 5. Jika rejected, karantina inventori
    if (result === 'rejected') {
      await tx.update(materialInventory)
        .set({ 
          isQuarantined: true,
          quarantineReason: `MIR #${inspectionId} ditolak: ${notes || 'tidak ada catatan'}`,
          quarantinedAt: new Date(),
          quarantinedBy: userId,
        })
        .where(eq(materialInventory.batchId, batch.id));
    }

    // 6. Catat audit trail
    await tx.insert(auditLogs).values({
      projectId,
      userId,
      action: 'SYNC_MIR_TO_BATCH',
      recordType: 'material_batch',
      recordId: batch.id,
      before: { 
        inspectionStatus: previousStatus,
        mirInspectionId: batch.mirInspectionId 
      },
      after: { 
        inspectionStatus: updatedBatch.inspectionStatus,
        mirInspectionId: inspection.id 
      },
      metadata: {
        inspectionId,
        result,
        notes,
      },
    });
  });
}

/**
 * Tautkan MIR ke batch material (dilakukan saat MIR dibuat)
 */
export async function linkMIRToBatch(params: {
  inspectionId: number;
  batchId: number;
  projectId: string;
  userId: string;
}): Promise<void> {
  const { inspectionId, batchId, projectId, userId } = params;

  await db.transaction(async (tx) => {
    // Verifikasi batch dalam proyek yang sama
    const batch = await tx.query.materialBatches.findFirst({
      where: and(
        eq(materialBatches.id, batchId),
        eq(materialBatches.projectId, projectId)
      ),
    });

    if (!batch) {
      throw new MIRBatchSyncError(
        'Batch material tidak ditemukan dalam proyek ini',
        'PROJECT_ISOLATION_VIOLATION'
      );
    }

    // Update batch
    await tx.update(materialBatches)
      .set({
        mirInspectionId: inspectionId,
        mirLinkedAt: new Date(),
        mirLinkedBy: userId,
        updatedAt: new Date(),
      })
      .where(eq(materialBatches.id, batchId));

    // Update inspeksi
    await tx.update(inspections)
      .set({
        relatedBatchId: batchId,
      })
      .where(eq(inspections.id, inspectionId));

    // Audit
    await tx.insert(auditLogs).values({
      projectId,
      userId,
      action: 'LINK_MIR_TO_BATCH',
      recordType: 'material_batch',
      recordId: batchId,
      before: { mirInspectionId: null },
      after: { mirInspectionId: inspectionId },
      metadata: { inspectionId, batchId },
    });
  });
}
```

```typescript
// artifacts/api-server/src/routes/materials.ts
import { Router } from 'express';
import { requireAuth, requireRole } from '../middlewares/auth';
import { linkMIRToBatch } from '../services/mir-batch-sync';
import { db } from '../lib/db';
import { materialBatches, inspections } from '../../lib/db/schema';
import { eq, and } from 'drizzle-orm';

const router = Router();

// POST /api/projects/:projectId/materials/batches/:batchId/link-mir
router.post(
  '/projects/:projectId/materials/batches/:batchId/link-mir',
  requireAuth,
  requireRole(['QA_ENGINEER', 'QA_ENGINEER_MADYA', 'QA_MANAGER']),
  async (req, res) => {
    const { projectId, batchId } = req.params;
    const { inspectionId } = req.body;
    const userId = req.auth.userId;

    try {
      await linkMIRToBatch({
        inspectionId,
        batchId: parseInt(batchId),
        projectId,
        userId,
      });

      res.status(200).json({ success: true });
    } catch (error) {
      if (error.code === 'PROJECT_ISOLATION_VIOLATION') {
        return res.status(403).json({ error: 'Akses ditolak' });
      }
      if (error.code === 'BATCH_NOT_LINKED') {
        return res.status(400).json({ error: error.message });
      }
      res.status(500).json({ error: 'Gagal menautkan MIR' });
    }
  }
);

// GET /api/projects/:projectId/materials/batches/:batchId/trace
// Tambahkan informasi MIR
router.get(
  '/projects/:projectId/materials/batches/:batchId/trace',
  requireAuth,
  async (req, res) => {
    const { projectId, batchId } = req.params;

    try {
      const batch = await db.query.materialBatches.findFirst({
        where: and(
          eq(materialBatches.id, parseInt(batchId)),
          eq(materialBatches.projectId, projectId)
        ),
        with: {
          materialType: true,
          mirInspection: {
            columns: ['id', 'status', 'approvedAt', 'approvedBy'],
            with: {
              approvedByUser: {
                columns: ['id', 'name', 'email'],
              },
            },
          },
          inventory: {
            with: {
              storageLocation: true,
            },
          },
          usageRecords: {
            with: {
              installedByUser: true,
            },
          },
        },
      });

      if (!batch) {
        return res.status(404).json({ error: 'Batch tidak ditemukan' });
      }

      res.json(batch);
    } catch (error) {
      res.status(500).json({ error: 'Gagal mengambil trace batch' });
    }
  }
);

export default router;
```

```typescript
// Integrasikan sinkronisasi ke endpoint approval/rejection MIR
import { syncMIRResultToBatch } from '../services/mir-batch-sync';

router.post(
  '/projects/:projectId/inspections/:inspectionId/resolve',
  requireAuth,
  requireRole(['QA_ENGINEER', 'QA_ENGINEER_MADYA', 'QA_MANAGER']),
  async (req, res) => {
    const { projectId, inspectionId } = req.params;
    const { result, notes } = req.body;
    const userId = req.auth.userId;

    try {
      // Update inspeksi (logika existing)
      const [updatedInspection] = await db.update(inspections)
        .set({
          status: result,
          resolvedAt: new Date(),
          resolvedBy: userId,
          resolutionNotes: notes,
        })
        .where(and(
          eq(inspections.id, parseInt(inspectionId)),
          eq(inspections.projectId, projectId)
        ))
        .returning();

      // Sinkronisasi ke batch jika ini MIR
      if (updatedInspection.type === 'MATERIAL_INSPECTION') {
        await syncMIRResultToBatch({
          inspectionId: parseInt(inspectionId),
          projectId,
          result,
          userId,
          notes,
        });
      }

      res.json(updatedInspection);
    } catch (error) {
      if (error.code === 'PROJECT_ISOLATION_VIOLATION') {
        return res.status(403).json({ error: 'Akses ditolak' });
      }
      res.status(500).json({ error: 'Gagal memproses MIR' });
    }
  }
);
```

```tsx
        projectId,
```
```typescript
        type: 'MATERIAL_INSPECTION',

        .send({ inspectionId: newMir.id });

      const audit = await db.query.auditLogs.findFirst({
        where: and(
          eq(auditLogs.recordType, 'material_batch'),
          eq(auditLogs.recordId, batch.id),

      expect(audit).toBeDefined();

      expect(audit).toBeDefined();
      expect(audit.before.inspectionStatus).toBe('pending');
      expect(audit.after.inspectionStatus).toBe('approved');
      expect(audit.metadata.inspectionId).toBe(mir.id);
      expect(audit.metadata.result).toBe('approved');
    });
  });

  describe('Transaction Rollback', () => {
```

## Ringkasan Implementasi

| Aspek | Implementasi |
|---|---|
| **Schema** | Tambah `mirInspectionId`, `mirLinkedAt`, `mirLinkedBy` di `materialBatches` |
| **Service** | `syncMIRResultToBatch()` dan `linkMIRToBatch()` dengan transaksi atomik |
| **API** | Endpoint link MIR, endpoint resolve MIR dengan sinkronisasi otomatis |
| **UI** | Tampilkan referensi MIR di batch trace, form untuk tautkan MIR saat create |
| **Audit** | Catat setiap perubahan status dan link dengan before/after |
| **Isolation** | Verifikasi batch dan MIR dalam proyek yang sama |
| **Rollback** | Transaksi memastikan semua atau tidak ada perubahan |
| **Tests** | Cover project isolation, status transitions, audit, rollback, RBAC |

Implementasi ini memastikan status MIR dan batch selalu sinkron, dengan audit trail lengkap dan isolasi proyek yang ketat.
